import { S3Client, PutObjectCommand } from '@aws-sdk/client-s3';
import { getSignedUrl } from '@aws-sdk/s3-request-presigner';

/** Generates a short-lived S3 PUT presigned URL for direct browser uploads */
export const handler = async (event: APIGatewayProxyEvent) => {
  const { fileName, fileSize, mimeType } = JSON.parse(event.body!);
  const s3Key = `users/${userId}/files/${fileId}/${fileName}`;

  const command = new PutObjectCommand({
    Bucket: process.env.FILES_BUCKET_NAME,
    Key: s3Key,
    ContentType: mimeType,
  });

  const uploadUrl = await getSignedUrl(s3Client, command, { expiresIn: 900 });
  return { statusCode: 200, body: JSON.stringify({ uploadUrl, fileId, s3Key }) };
};

Senior Full-Stack Design

Production-Grade AWS & Angular Architecture

Built with strict adherence to AWS Well-Architected and modern Angular 22 best practices.

bolt

Angular 22 Signals State

Fine-grained reactive state management using signal, computed, and effect with zoneless readiness and lazy-loaded routing bundles.

cloud_sync

Zero-Proxy Direct S3 Uploads

3-Phase presigned URL protocol. Client streams files directly to S3 via PUT requests, eliminating Lambda compute execution time and memory limits for large uploads.

table_chart

Single-Table DynamoDB

Optimized single-table design with composite partition/sort keys (PK/SK) and Global Secondary Indexes (GSI1/GSI2) for sub-10ms queries.

security

Cognito JWT Security

API Gateway JWT Authorizer validation, short-lived tokens, cryptographic token revocation via RevokeTokenCommand, and secure sessionStorage isolation.

code_blocks

100% TypeScript AWS CDK

Repeatable Infrastructure as Code (IaC) defining S3 buckets, DynamoDB tables, Lambda ESM handlers, and least-privilege IAM policies.

verified

Comprehensive Testing Suite

Unit testing with Vitest for backend Lambda handlers and Angular TestBed for components, guards, and services with 100% route verification.
